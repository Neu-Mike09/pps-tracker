    80→async function safeJsonResponse(res: Response): Promise<Record<string, unknown>> {
    81→  const contentType = res.headers.get("content-type") || "";
    82→  const text = await res.text();
    83→  if (contentType.includes("application/json") || text.trim().startsWith("{") || text.trim().startsWith("[")) {
    84→    try { return JSON.parse(text); } catch {}
    85→  }
    86→  const preview = text.slice(0, 200).trim() || "(empty response)";
    87→  return { error: `Server returned a non-JSON response (${res.status} ${res.statusText}). ${preview}` };
    88→}

    90→async function fetchWithTimeout(url: string, options: RequestInit, ms: number): Promise<Response> {
    91→  const controller = new AbortController();
    92→  const timeoutId = setTimeout(() => controller.abort(), ms);
    93→  try {
    94→    return await fetch(url, { ...options, signal: controller.signal });
    95→  } catch (e) {
    96→    if (e instanceof DOMException && e.name === "AbortError") {
    97→      throw new Error("Request timed out — the AI took too long to respond. Please try again.");
    98→    }
    99→    throw e;
   100→  } finally {
   101→    clearTimeout(timeoutId);
   102→  }
   103→}

   105→const toLocalDateTime = (iso: string | null): string | null => {
   106→  if (!iso) return null;
   107→  const d = new Date(iso);
   108→  if (isNaN(d.getTime())) return null;
   109→  const pad = (n: number) => String(n).padStart(2, "0");
   110→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   111→};
   112→const toLocalDate = (iso: string | null): string | null => {
   113→  if (!iso) return null;
   114→  const d = new Date(iso);
   115→  if (isNaN(d.getTime())) return null;
   116→  return d.toISOString().slice(0, 10);
   117→};
