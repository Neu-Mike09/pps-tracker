   100→const isImageFile = (path: string): boolean => {
   101→  const ext = path.split(".").pop()?.toLowerCase() || "";

   104→
   105→const fmtDate = (s: string | null): string => {
   106→  if (!s) return "";
   107→  const d = new Date(s);
   108→  if (isNaN(d.getTime())) return "";
   109→  return d.toLocaleDateString("en-US", { year: "numeric", month: "short", day: "numeric" });
   110→};
   111→
   112→// Convert "16:00" (24-hour HH:MM) to "4:00 PM" (12-hour)
   113→const fmtTime12 = (time24: string | null): string => {
   114→  if (!time24) return "";
   115→  const [h, m] = time24.split(":").map(Number);
   116→  if (isNaN(h)) return time24;
   117→  const period = h >= 12 ? "PM" : "AM";
   118→  const h12 = h % 12 || 12;
   119→  return `${h12}:${String(m || 0).padStart(2, "0")} ${period}`;
   120→};

  1028→function toLocalDateTimeInput(iso: string): string {
  1029→  const d = new Date(iso);
  1030→  if (isNaN(d.getTime())) return "";
  1031→  const pad = (n: number) => String(n).padStart(2, "0");
  1032→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
  1033→}
